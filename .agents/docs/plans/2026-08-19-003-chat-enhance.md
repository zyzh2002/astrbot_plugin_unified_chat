/// FNV-1a 64-bit dedup hash, lowercase hex, deterministic.
#[pyfunction]
fn hash_dedup(text: &str) -> String {
    let mut h: u64 = 0xcbf29ce484222325;
    for b in text.as_bytes() {
        h ^= *b as u64;
        h = h.wrapping_mul(0x100000001b3);
    }
    format!("{h:016x}")
}

# fallback.py
def hash_dedup(text: str) -> str:
    """FNV-1a 64-bit hash, lowercase hex (bit-identical to Rust impl)."""
    h = 0xCBF29CE484222325
    for b in text.encode("utf-8"):
        h ^= b
        h = (h * 0x100000001B3) & 0xFFFFFFFFFFFFFFFF
    return f"{h:016x}"

```python
from collections import deque
from unified_chat.native import chunk_text
from unified_chat.utils.hashing import dedup_hash

class ChatService:
    MAX_SESSION_HISTORY = 50
    MAX_CONTEXT_CHARS = 4000
    DEDUP_WINDOW = 20
    SNIPPET_LEN = 120

    def __init__(self):
        self._buffers: dict[str, deque] = {}
        self._seen: dict[str, deque] = {}

    @staticmethod
    def is_command(text: str) -> bool:
        t = text.strip()
        return not t or t.startswith("/")

    def should_process(self, event) -> bool:
        return not self.is_command(getattr(event, "message_str", ""))

    @staticmethod
    def hash_of(text: str) -> str:
        return dedup_hash(text)

    def seen_hash(self, session: str, h: str) -> bool:
        return h in self._seen.get(session, ())

    def remember_hash(self, session: str, h: str) -> None:
        q = self._seen.setdefault(session, deque(maxlen=self.DEDUP_WINDOW))
        q.append(h)

    def record(self, event) -> None:
        session = event.unified_msg_origin
        text = getattr(event, "message_str", "")
        sender = ""
        try: sender = event.get_sender_name() or ""
        except Exception: pass
        snippet = text[:self.SNIPPET_LEN]
        try:
            chunks = chunk_text(text, self.SNIPPET_LEN, 0)
            if chunks: snippet = chunks[0]
        except Exception: pass
        buf = self._buffers.setdefault(session, deque(maxlen=self.MAX_SESSION_HISTORY))
        buf.append((sender, snippet))

    def social_context(self, event) -> str:
        if event.is_private_chat():
            return ""
        buf = self._buffers.get(event.unified_msg_origin)
        if not buf:
            return ""
        senders = []
        for sender, _ in buf:
            if sender and sender not in senders:
                senders.append(sender)
        senders = senders[-10:]
        lines = [f"Recently active: {', '.join(senders)}"]
        for sender, snippet in list(buf)[-5:]:
            who = sender or "?"
            lines.append(f"{who}: {snippet}")
        ctx = "\n".join(lines)
        return ctx[:self.MAX_CONTEXT_CHARS]
```

async def inject_social_context(event, req, config, chat_service) -> None:
    if not config.enable_conversation_enhance:
        return
    try:
        social = chat_service.social_context(event)
        if not social:
            return
        contexts = getattr(req, "contexts", None)
        if contexts is None:
            contexts = []
            req.contexts = contexts
        contexts.append({"role": "system", "content": social})
    except Exception:
        with contextlib.suppress(Exception):
            from astrbot.api import logger

```python
# unified_chat/core/pipeline.py
import asyncio
import contextlib

class MessagePipeline:
    def __init__(self, config, chat_service):
        self.config = config
        self.chat_service = chat_service

    async def process(self, event) -> None:
        if not self.config.enable_conversation_enhance:
            return
        try:
            if not self.chat_service.should_process(event):
                return
            session = event.unified_msg_origin
            h = self.chat_service.hash_of(event.message_str)
            if self.chat_service.seen_hash(session, h):
                return
            self.chat_service.remember_hash(session, h)
            self.chat_service.record(event)
            self._spawn_background(event)
        except Exception:
            self._log_error("process")

    def _spawn_background(self, event) -> None:
        task = asyncio.create_task(self._after_stages(event), name="unified_chat_pipeline")
        task.add_done_callback(self._log_done)

    async def _after_stages(self, event) -> None:
        # Extension point: memory (004), learning (005)
        return None

    def _log_done(self, task) -> None:
        with contextlib.suppress(Exception):
            exc = task.exception()
            if exc is not None:
                self._log_error(f"background: {exc}")

    @staticmethod
    def _log_error(msg: str) -> None:
        with contextlib.suppress(Exception):
            from astrbot.api import logger
            logger.error(f"[unified_chat] pipeline {msg}", exc_info=True)
```

lifecycle: `on_load` after RagService create ChatService+Pipeline; `handle_message` try/except calls `await self._pipeline.process(event)`; on_llm_request adds `inject_social_context` call after RAG injection.
