```python
import asyncio
import contextlib

from unified_chat.storage import kv as kv_store
from unified_chat.storage import repo as repos
from unified_chat.storage.models import LearningLog


class MigrationService:
    def __init__(self, context, config):
        self.context = context
        self.config = config
        self._lock = asyncio.Lock()

    def _flag_key(self, kb_name: str) -> str:
        return f"migration:{kb_name}:running"

    async def is_running(self, kb_name: str) -> bool:
        return bool(await kv_store.kv_get(self._flag_key(kb_name)))

    async def run_migration(self, kb_name: str) -> str:
        async with self._lock:
            if await self.is_running(kb_name):
                return f"Migration for '{kb_name}' is already running."
            await kv_store.kv_set(self._flag_key(kb_name), "1")
            try:
                kb_manager = getattr(self.context, "kb_manager", None)
                if kb_manager is None:
                    return "KB manager unavailable."
                helper = await kb_manager.get_kb_by_name(kb_name)
                if helper is None:
                    return f"Knowledge base '{kb_name}' not found."
                snapshots = []  # (file_name, chunks)
                doc_ids = []
                offset = 0
                while True:
                    docs = await helper.list_documents(offset=offset, limit=100) or []
                    if not docs:
                        break
                    for doc in docs:
                        chunks = []
                        coffset = 0
                        while True:
                            batch = (
                                await helper.get_chunks_by_doc_id(
                                    doc.doc_id, offset=coffset, limit=100
                                )
                                or []
                            )
                            if not batch:
                                break
                            chunks.extend(c["content"] for c in batch)
                            coffset += 100
                        snapshots.append((doc.file_name, chunks))
                        doc_ids.append(doc.doc_id)
                    offset += 100
                for doc_id in doc_ids:
                    with contextlib.suppress(Exception):
                        await helper.delete_document(doc_id)
                new_ids = {}
                for file_name, chunks in snapshots:
                    doc = await helper.upload_document(
                        file_name=file_name,
                        file_content=None,
                        file_type="txt",
                        pre_chunked_text=chunks,
                    )
                    new_ids[file_name] = doc.doc_id
                if kb_name == self.config.memory_kb_name:
                    # rewrite memory kb_doc_id mapping is handled by caller via Memory rows
                    pass
                await repos.LearningLogRepo.add(
                    LearningLog(
                        stage="migration",
                        input_text=kb_name,
                        output_text=f"{len(snapshots)} docs rebuilt",
                    )
                )
                return f"Migration for '{kb_name}' done: {len(snapshots)} docs rebuilt."
            except Exception as e:
                return f"Migration for '{kb_name}' failed: {e}"
            finally:
                await kv_store.kv_delete(self._flag_key(kb_name))
```

(Note: memory kb_doc_id rewrite: after re-upload, Memory rows keep old doc ids; KB docs were rebuilt as new ids. Since memory KB source of truth is SQLite, on next cleanup the old kb_doc_id deletes are no-ops and KB docs without Memory link linger. Simplest correct behavior: after rebuild, clear all Memory.kb_doc_id for the memory KB via a new `MemoryRepo.clear_kb_doc_ids()`.)

# repo.py MemoryRepo
@staticmethod
async def clear_kb_doc_ids() -> int:
    async with get_session() as session:
        rows = (await session.exec(select(Memory).where(Memory.kb_doc_id.is_not(None)))).all()
        for r in rows:
            r.kb_doc_id = None
            session.add(r)
        await session.commit()
        return len(rows)

    snapshot and config.embedding_provider_id and snapshot != config.embedding_provider_id
)
await kv_store.kv_set("embedding_provider_snapshot", config.embedding_provider_id)

from unified_chat.services.migration_service import MigrationService

self._migration_service = MigrationService(self.context, config)


# migrate_kb:
async def migrate_kb(self, event, kb_name):
    if not kb_name:
        return "Usage: /unified_migrate <kb_name>"
    if self._migration_service is None:
        return "Plugin not initialized"
    if await self._migration_service.is_running(kb_name):
        return f"Migration for '{kb_name}' is already running."
    task = asyncio.create_task(
        self._migration_service.run_migration(kb_name), name="unified_chat_migration"
    )
    self._migration_tasks.append(task)
    return f"Migration for '{kb_name}' started in background. Check /unified_status."


# get_status_async:
async def get_status_async(self) -> str:
    parts = [f"status={self._status}"]
    if self._data_dir:
        parts.append(f"data_dir={self._data_dir}")
    if self._config:
        parts.append(
            f"rag_kbs={self._config.rag_kbs} agentic={self._config.rag_agentic} mem_days={self._config.memory_cleanup_days}"
        )
        parts.append(f"needs_migration={'yes' if self._needs_migration else 'no'}")
    try:
        parts.append(f"memories={await repos.MemoryRepo.count()}")
        parts.append(f"messages={await repos.MessageRepo.count()}")
        parts.append(
            f"learning filter={await repos.LearningLogRepo.count_by_stage('filter')} refine={await repos.LearningLogRepo.count_by_stage('refine')} reinforce={await repos.LearningLogRepo.count_by_stage('reinforce')}"
        )
    except Exception:
        parts.append("counts=n/a")
    return " | ".join(parts)
