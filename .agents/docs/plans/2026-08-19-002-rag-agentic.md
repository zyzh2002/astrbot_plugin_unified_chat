> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Inject a plugin-scoped `FunctionTool` (`unified_chat_kb_query`) into `ProviderRequest.func_tool` on `on_llm_request`, agentic-only, bound to plugin `rag_kbs`.

**Architecture:** `RagService(context).build_kb_tool(kb_names)` lazily imports AstrBot tool classes and returns a `FunctionTool` subclass whose `call` delegates to `context.kb_manager.retrieve`. `core/hooks.inject_kb_tool` gates on config and appends to `req.func_tool` (creating `ToolSet` when `None`). Lifecycle wires both.

**Tech Stack:** Python 3.12, pydantic v2 (dev dep; AstrBot provides at runtime), AstrBot `FunctionTool`/`ToolSet`/`ProviderRequest`.

**Spec:** `.agents/docs/specs/002-rag-agentic.md`

- AstrBot `>=4.27.3,<5.0.0`; verified API: `req.func_tool: ToolSet | None`, `ToolSet.add_tool(FunctionTool)`, overridden `call(ContextWrapper, **kwargs) -> str | mcp CallToolResult`, `kb_manager.retrieve(query, kb_names, top_k_fusion, top_m_final) -> dict | None`
- Agentic only; never raise from hook; lazy astrbot imports inside methods
- No global singletons; `RagService` per lifecycle

```python
# tests/conftest.py
import sys, types
from pydantic import Field
from pydantic.dataclasses import dataclass


def _install_astrbot_tool_stub():
    if "astrbot.core.agent.tool" in sys.modules:
        return
    pkg = types.ModuleType("astrbot.core.agent.tool")

    @dataclass
    class FunctionTool:
        name: str
        description: str
        parameters: dict = Field(default_factory=dict)
        handler: object = None
        active: bool = True

        async def call(self, context, **kwargs):
            raise NotImplementedError(
                "FunctionTool.call() must be implemented by subclasses or set a handler."
            )

    @dataclass
    class ToolSet:
        tools: list = Field(default_factory=list)

        def add_tool(self, tool):
            for i, existing in enumerate(self.tools):
                if existing.name == tool.name:
                    existing_active = bool(getattr(existing, "active", True))
                    new_active = bool(getattr(tool, "active", True))
                    if new_active or not existing_active:
                        self.tools[i] = tool
                    return
            self.tools.append(tool)

        def get_tool(self, name):
            for tool in self.tools:
                if tool.name == name:
                    return tool
            return None

    pkg.FunctionTool = FunctionTool
    pkg.ToolSet = ToolSet
    sys.modules["astrbot.core.agent.tool"] = pkg


_install_astrbot_tool_stub()
```

```python
# unified_chat/services/rag_service.py
TOOL_NAME = "unified_chat_kb_query"


class RagService:
    def __init__(self, context):
        self.context = context

    def build_kb_tool(self, kb_names, top_m_final=5):
        if not kb_names:
            return None
        kb_manager = getattr(self.context, "kb_manager", None)
        if kb_manager is None:
            return None
        try:
            from astrbot.core.agent.tool import FunctionTool
            from pydantic import Field
            from pydantic.dataclasses import dataclass
        except Exception:
            return None

        bound_names = list(kb_names)

        @dataclass
        class UnifiedChatKbQueryTool(FunctionTool):
            name: str = TOOL_NAME
            description: str = (
                "Query the plugin-configured knowledge base(s) for facts, "
                "background knowledge or previously indexed content. "
                "Only send a concise keyword query."
            )
            parameters: dict = Field(
                default_factory=lambda: {
                    "type": "object",
                    "properties": {
                        "query": {"type": "string", "description": "A concise keyword query."}
                    },
                    "required": ["query"],
                }
            )

            async def call(self, context, **kwargs):
                query = str(kwargs.get("query", "")).strip()
                if not query:
                    return "error: Query parameter is empty."
                try:
                    result = await kb_manager.retrieve(
                        query=query,
                        kb_names=bound_names,
                        top_k_fusion=20,
                        top_m_final=top_m_final,
                    )
                except Exception as e:
                    return f"error: {e}"
                if not result or not isinstance(result, dict):
                    return "No relevant knowledge found."
                text = str(result.get("context_text", "")).strip()
                return text or "No relevant knowledge found."

        return UnifiedChatKbQueryTool()
```

```python
# unified_chat/core/hooks.py
import contextlib


async def inject_kb_tool(event, req, config, rag_service) -> None:
    if not config.rag_agentic or not config.rag_kbs:
        return
    try:
        tool = rag_service.build_kb_tool(config.rag_kbs)
        if tool is None:
            return
        func_tool = getattr(req, "func_tool", None)
        if func_tool is None:
            from astrbot.core.agent.tool import ToolSet

            func_tool = ToolSet()
            req.func_tool = func_tool
        if func_tool.get_tool(tool.name) is not None:
            return
        func_tool.add_tool(tool)
    except Exception:
        with contextlib.suppress(Exception):
            from astrbot.api import logger

            logger.error("[unified_chat] inject_kb_tool failed", exc_info=True)
```
